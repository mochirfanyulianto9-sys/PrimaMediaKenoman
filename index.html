Generator Script Konfigurasi MikroTik (RouterOS v6 & v7) & Mini API Bridge

1. Script MikroTik RouterOS v7 (Aktifkan REST API HTTP/HTTPS + IP Cloud DDNS) 
/ip service set www disabled=no port=8080
/ip service set www-ssl disabled=no port=443
/ip cloud set ddns-enabled=yes
/user group add name=noc-api policy=read,write,api,rest-api,winbox
/user add name=noc-primamedia password=GantiPasswordKuat2026 group=noc-api
/ip firewall filter add chain=input protocol=tcp dst-port=443,8080 action=accept place-before=0 comment="Allow NOC Prima Media REST API"
2. Script MikroTik RouterOS v6 (Aktifkan API Port 8728 / API-SSL 8729 + Firewall) 
/ip service set api disabled=no port=8728
/ip service set api-ssl disabled=no port=8729
/ip cloud set ddns-enabled=yes
/user group add name=noc-v6 policy=read,write,api,winbox
/user add name=noc-primamedia password=GantiPasswordKuat2026 group=noc-v6
/ip firewall filter add chain=input protocol=tcp dst-port=8728,8729 action=accept place-before=0 comment="Allow NOC Prima Media RouterOS v6 API"
3. Mini Bridge 1-File Node.js (bridge-mikrotik.js) — Jembatan WebSocket ke Port 8728 (v6 & v7)

Karena browser tidak bisa membuka socket TCP mentah ke port 8728 RouterOS v6 secara langsung, jalankan 1 file kecil ini di VPS/PC NOC (`npm i ws routeros-client` lalu `node bridge-mikrotik.js`):

const { WebSocketServer } = require('ws');
const { RouterOSAPI } = require('routeros-client');

const wss = new WebSocketServer({ port: 8787 });
console.log('Prima media MikroTik Bridge aktif di ws://0.0.0.0:8787');

wss.on('connection', (ws) => {
  ws.on('message', async (raw) => {
    const req = JSON.parse(raw);
    const api = new RouterOSAPI({ host: req.host, port: Number(req.port || 8728), user: req.user, password: req.pass, timeout: 5 });
    try {
      const client = await api.connect();
      if (req.action === 'sync') {
        const [res, secrets, active, eth] = await Promise.all([
          client.menu('/system/resource').get(),
          client.menu('/ppp/secret').get(),
          client.menu('/ppp/active').get(),
          client.menu('/interface').where('type', 'ether').get()
        ]);
        ws.send(JSON.stringify({ ok: true, action: 'sync', resource: res[0], secrets, active, interfaces: eth }));
      } else if (req.action === 'exec') {
        const out = await client.menu(req.path).exec(req.cmd, req.params || {});
        ws.send(JSON.stringify({ ok: true, action: 'exec', data: out }));
      }
      await api.close();
    } catch (err) {
      ws.send(JSON.stringify({ ok: false, error: err.message }));
    }
  });
});
